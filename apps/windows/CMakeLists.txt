cmake_minimum_required(VERSION 3.24)
project(SaySpanWindows VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive- /EHsc)
endif()
enable_testing()
add_library(editor STATIC src/editor.cpp)
target_include_directories(editor PUBLIC src)
add_executable(editor_tests tests/editor_tests.cpp)
target_link_libraries(editor_tests PRIVATE editor)
add_test(NAME editor COMMAND editor_tests)
if(NOT WIN32)
  return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "SaySpan supports only 64-bit x64 and ARM64 targets")
endif()
enable_language(RC)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../macos/Resources/Branding.json" BRANDING_JSON)
string(JSON SAYSPAN_NAME GET "${BRANDING_JSON}" name)
configure_file(src/branding.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/branding.h" @ONLY)
configure_file(resources/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc" @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
set(SAYSPAN_DEPENDENCIES "" CACHE PATH "Prefix built by tools/dependencies.py")
find_path(RIME_INCLUDE_DIR rime_api.h PATHS "${SAYSPAN_DEPENDENCIES}/include" REQUIRED NO_DEFAULT_PATH)
file(GLOB_RECURSE RIME_DLL "${SAYSPAN_DEPENDENCIES}/*rime.dll")
list(LENGTH RIME_DLL RIME_DLL_COUNT)
if(NOT RIME_DLL_COUNT EQUAL 1)
  message(FATAL_ERROR "Expected exactly one rime.dll in the dependency prefix")
endif()
add_compile_definitions(UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
add_library(runtime STATIC src/rime_engine.cpp)
target_include_directories(runtime PRIVATE "${RIME_INCLUDE_DIR}")
target_link_libraries(runtime PUBLIC editor ole32 shell32 user32 uuid advapi32)
add_library(SaySpanTip SHARED src/service.cpp src/registration.cpp src/panel.cpp src/bridge.cpp src/exports.def "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc")
target_link_libraries(SaySpanTip PRIVATE runtime uuid oleaut32 advapi32 gdi32)
add_executable(SaySpanRegister src/register_main.cpp)
target_include_directories(SaySpanRegister PRIVATE src)
target_link_libraries(SaySpanRegister PRIVATE ole32 uuid)
add_executable(runtime_tests tests/runtime_tests.cpp)
target_link_libraries(runtime_tests PRIVATE runtime)
add_executable(host_tests tests/host_tests.cpp)
target_include_directories(host_tests PRIVATE src)
target_link_libraries(host_tests PRIVATE ole32 uuid user32)
foreach(target SaySpanTip SaySpanRegister runtime_tests editor_tests host_tests)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_custom_command(TARGET runtime_tests POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RIME_DLL}" "$<TARGET_FILE_DIR:runtime_tests>/rime.dll")
